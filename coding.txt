    <script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js" integrity="sha256-20nQCchB9co0qIjJZRGuk2/Z9VM+kNiyxNV1lvTlZBo=" crossorigin=""></script>
    <script>
        const map = L.map('map', { zoomControl: false, attributionControl: false });
        
        const googleSatelliteLayer = L.tileLayer('https://{s}.google.com/vt/lyrs=s&x={x}&y={y}&z={z}', { maxZoom: 20, subdomains: ['mt0', 'mt1', 'mt2', 'mt3'] }).addTo(map);
        const osmLayer = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 19 });
        
        let currentBasemap = 'satellite';
        const cleanName = str => str ? str.replace(/kabupaten\s*/gi, '').trim() : "-";

        const layers = {
            Soppeng: L.layerGroup().addTo(map),
            Bone: L.layerGroup().addTo(map)
        };

        const kabupatenDataConfig = {
            "Soppeng": { nama: "Soppeng", luas: "1.500 km²", penduduk: "238.220 Jiwa", style: { color: '#0284c7', weight: 2, opacity: 1, fillColor: '#0284c7', fillOpacity: 0.85 } },
            "Bone": { nama: "Bone", luas: "4.559 km²", penduduk: "801.775 Jiwa", style: { color: '#d97706', weight: 2, opacity: 1, fillColor: '#d97706', fillOpacity: 0.85 } }
        };

        function toggleKabupatenLayer(kab, isChecked) {
            isChecked ? map.addLayer(layers[kab]) : map.removeLayer(layers[kab]);
        }

        // Fungsi Tunggal untuk Mengubah Card Info (Mencegah Redundansi Kode)
        function updateDetailCard(type, data) {
            toggleSidebar(false);
            const isKab = type === 'KABUPATEN';
            
            document.getElementById('cardHeaderTitle').innerText = isKab ? "INFORMASI KABUPATEN" : "DETAIL WILAYAH PERENCANAAN";
            document.getElementById('labelCardName').innerText = isKab ? "Kabupaten:" : "Nama WP:";
            document.getElementById('cardName').innerText = cleanName(data.nama);

            document.getElementById('rowKabLuas').style.display = isKab ? 'flex' : 'none';
            document.getElementById('rowKabPenduduk').style.display = isKab ? 'flex' : 'none';
            document.getElementById('rowStatusWP').style.display = isKab ? 'none' : 'flex';
            document.getElementById('rowProgres').style.display = isKab ? 'none' : 'flex';

            if (isKab) {
                document.getElementById('cardKabLuas').innerText = data.luas;
                document.getElementById('cardKabPenduduk').innerText = data.penduduk;
            } else {
                document.getElementById('cardStatus').innerText = data.status || "Penyusunan Matek";
                document.getElementById('cardProgres').innerText = data.progres || "Bantek TA 2026";
            }
            document.getElementById('detailCard').classList.add('visible');
        }

        function setupKabupatenEvents(feature, layer, configKey) {
            const config = kabupatenDataConfig[configKey];
            const props = feature.properties || {};
            const namaClean = cleanName(props.KABUPATEN || props.NAMOBJ || config.nama);
            const luas = props.LUAS_KM2 ? props.LUAS_KM2 + " km²" : config.luas;
            const penduduk = props.PENDUDUK ? props.PENDUDUK.toLocaleString('id-ID') + " Jiwa" : config.penduduk;

            layer.on({
                mouseover: e => e.target.setStyle({ fillOpacity: 0.35, weight: 3, color: '#ffffff' }),
                mouseout: e => e.target.setStyle(config.style),
                click: e => {
                    if (window.innerWidth <= 768) {
                        layer.bindPopup(`<div class="mobile-popup-card"><h4>Kabupaten ${namaClean}</h4><p><b>Total Luas:</b> ${luas}</p><p><b>Total Penduduk:</b> ${penduduk}</p></div>`).openPopup(e.latlng);
                    } else {
                        updateDetailCard('KABUPATEN', { nama: namaClean, luas, penduduk });
                    }
                }
            });
        }

        // Fetch Data GeoJSON
        ['Soppeng', 'Bone'].forEach(kab => {
            fetch(`${kab}.geojson`)
                .then(res => res.json())
                .then(data => {
                    L.geoJSON(data, {
                        style: kabupatenDataConfig[kab].style,
                        onEachFeature: (feature, layer) => setupKabupatenEvents(feature, layer, kab)
                    }).addTo(layers[kab]);
                }).catch(err => console.error(`Error ${kab} GeoJSON:`, err));
        });

        // Data Titik Pulse
        const locationsData = [
            { name: "RDTR Perkotaan Cangadi", kabupaten: "Kabupaten Soppeng", status: "Penyusunan Matek", progres: "Bantek TA 2026", lat: -4.372587, lng: 119.950761, zoom: 13 },
            { name: "RDTR Perkotaan Donri-Donri", kabupaten: "Kabupaten Soppeng", status: "Penyusunan Matek", progres: "Bantek TA 2026", lat: -4.272618, lng: 119.870246, zoom: 13 },
            { name: "RDTR Kws. Taccipi Kec. Ulaweng", kabupaten: "Kabupaten Bone", status: "Penyusunan Matek", progres: "Bantek TA 2026", lat: -4.506518, lng: 120.185027, zoom: 13 }
        ];

        const pulseIcon = L.divIcon({ className: '', html: '<div class="pulse-marker-container"><div class="pulse-ring"></div><div class="pulse-dot"></div></div>', iconSize: [32, 32], iconAnchor: [16, 16] });
        const markerMap = [], latLngs = [];
        const wpContainer = document.getElementById('wpListContainer');

        locationsData.forEach((loc, index) => {
            // Generasi Dinamis Item Sidebar List
            const li = document.createElement('li');
            li.id = `item-${index}`;
            li.onclick = () => selectLocation(index);
            li.innerHTML = `<div class="wp-title">📍 ${loc.name.replace("RDTR ", "")}</div><div class="wp-desc">${loc.kabupaten}</div>`;
            wpContainer.appendChild(li);

            // Marker Map
            const marker = L.marker([loc.lat, loc.lng], { icon: pulseIcon }).addTo(map);
            marker.bindPopup(`<div class="mobile-popup-card"><h4>${loc.name}</h4><div class="subtitle">${loc.kabupaten}</div><p><b>Status:</b> ${loc.status}</p><p><b>Progres:</b> ${loc.progres}</p></div>`, { maxWidth: 260 });
            marker.on('click', () => selectLocation(index));

            markerMap.push(marker);
            latLngs.push([loc.lat, loc.lng]);
        });

        const initialBounds = L.latLngBounds(latLngs);
        map.fitBounds(initialBounds, { padding: [50, 50], maxZoom: 10 });

        function selectLocation(index) {
            const data = locationsData[index];
            if (window.innerWidth <= 768) {
                map.setView([data.lat, data.lng], data.zoom, { animate: true });
                markerMap[index].openPopup();
            } else {
                markerMap[index].closePopup();
                map.setView([data.lat, data.lng], data.zoom, { animate: true });
                updateDetailCard('WP', { nama: `${data.name} (${data.kabupaten})`, status: data.status, progres: data.progres });

                document.querySelectorAll('.wp-list li').forEach((el, i) => el.classList.toggle('active', i === index));
            }
        }

        function toggleSidebar(collapse) {
            document.getElementById('sidebarPanel').classList.toggle('collapsed', collapse);
            document.getElementById('openSidebarBtn').classList.toggle('visible', collapse);
        }

        function toggleBasemap() {
            const isSat = currentBasemap === 'satellite';
            map.removeLayer(isSat ? googleSatelliteLayer : osmLayer);
            (isSat ? osmLayer : googleSatelliteLayer).addTo(map);
            currentBasemap = isSat ? 'osm' : 'satellite';

            document.getElementById('globeBtn').classList.toggle('osm-mode', isSat);
            document.getElementById('globeLabel').textContent = isSat ? "OpenStreetMap (OSM)" : "Google Satellite";
        }

        function resetMapView() {
            map.closePopup();
            map.fitBounds(initialBounds, { padding: [50, 50], maxZoom: 10, animate: true });
            document.getElementById('detailCard').classList.remove('visible');
            document.querySelectorAll('.wp-list li').forEach(el => el.classList.remove('active'));
        }
    </script>
